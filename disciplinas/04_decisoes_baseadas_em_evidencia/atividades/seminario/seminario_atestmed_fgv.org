\begin{frame}[t]{O problema público}
\vspace{0.15cm}
{\large\textbf{Capacidade pericial limitada diante de demanda heterogênea}}\par\vspace{0.42cm}
\begin{itemize}\setlength\itemsep{0.45em}
\item Fila superior a 90 dias em determinadas localidades.
\item Parte dos requerimentos possui duração inferior ao tempo de espera.
\item Perícia presencial é recurso escasso.
\item Consequências: demora, judicialização, pagamentos prolongados e sobrecarga administrativa.
\end{itemize}
\vfill
\FGVCallout{\textbf{Pergunta-chave:} como reservar a perícia presencial para os casos em que ela realmente produz maior valor decisório?}
\end{frame}
#+END_EXPORT

\begin{frame}[t]{Problema não é solução}
\begin{columns}[T,onlytextwidth]
\column{0.49\textwidth}
\begin{block}{Problema público}\small
Capacidade pericial limitada distribuída sem priorização suficiente por risco, complexidade e valor esperado da perícia presencial.
\end{block}
\vspace{0.12cm}
\begin{block}{Efeitos observados}\small
Fila heterogênea, demora excessiva, judicialização, pagamentos prolongados e sobrecarga da Perícia Médica Federal.
\end{block}
\column{0.49\textwidth}
\begin{block}{Alternativas possíveis}\small
\begin{itemize}\setlength\itemsep{0.28em}
\item Manter o status quo.
\item Reforçar ou redistribuir a capacidade pericial.
\item Fazer mutirões e ampliar teleperícia.
\item Usar ATESTMED com regra fixa.
\item Usar ATESTMED estatístico com auditoria por risco.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\begin{frame}[t]{Modelo racionalista aplicado ao ATESTMED}

{\large\textbf{Modelo racionalista de análise de políticas públicas}}\par\vspace{0.34cm}
\begin{enumerate}\setlength\itemsep{0.35em}

\item Definição do objetivo.

\end{block}

\begin{frame}[t]{Modelo estatístico: duração, risco e contexto laboral}
\begin{block}{Estágio 1 - duração esperada do afastamento}
\[
Q_{\tau}(D \mid \text{CID, idade, sexo, território, atividade, exposição, histórico laboral})
\]
\small Produz percentis condicionais: P50, P75, P90 e P95.
\end{block}
\vspace{0.10cm}
\begin{block}{Estágio 2 - risco de divergência ou necessidade de revisão}
\[
P(\text{divergência}=1 \mid X)
\]
\small $X$ inclui duração solicitada, qualidade documental, atividade/CBO, exigências funcionais, tempo na ocupação, histórico de afastamentos e histórico decisório.
\end{block}
\vfill
\FGVCallout{\small A atividade laboral contextualiza a limitação funcional e melhora a comparação estatística; não funciona como rótulo automático de deferimento ou indeferimento.}
\end{frame}
#+END_EXPORT

\begin{frame}[t]{Fluxo completo: do atestado à decisão}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{1. Entrada e validação}\small
Requerimento, atestado digital, CID, dias solicitados, atividade laboral, histórico laboral e dados administrativos.
\end{block}
\begin{block}{2. Tratamento automatizado}\small
Autenticidade, completude, padronização de variáveis, cruzamentos autorizados e identificação de inconsistências.
\end{block}
\begin{block}{3. Contextualização estatística}\small
Grupo de referência, duração esperada, risco de divergência, impacto potencial e incerteza.
\end{block}
\column{0.48\textwidth}
\begin{block}{4. Roteamento da decisão}\small
\textbf{Baixo risco}: deferimento documental automatizado + auditoria aleatória.\par\vspace{0.25em}
\textbf{Médio risco}: auditoria documental humana.\par\vspace{0.25em}
\textbf{Alto risco/incerteza}: perícia presencial prioritária.
\end{block}
\begin{block}{5. Controle e aprendizado}\small
Auditoria independente, registro de divergências, monitoramento de vieses, recalibração e revisão periódica das regras.
\end{block}
\end{columns}
\end{frame}

\begin{frame}[t]{Caso Rafael: decisão em três passos}
\begin{block}{Passo 1 - integridade documental}\small
Assinatura digital válida, emissor identificável, campos completos e consistência entre dias solicitados, atividade declarada e requisitos administrativos.
\end{block}
\vspace{0.15cm}
\begin{block}{Passo 2 - avaliação estatística contextualizada}\small
O sistema estima o percentil dos 45 dias e calcula $P(\text{divergência})$, incorporando duração, documentação, atividade laboral, histórico laboral e incerteza.
\end{block}
\vspace{0.15cm}
\begin{block}{Passo 3 - decisão de fluxo}\small
\textbf{Baixo risco}: deferimento documental automatizado, se permitido pela regra vigente, com auditoria aleatória posterior.\par\vspace{0.25em}
\textbf{Risco intermediário}: auditoria documental humana.\par\vspace{0.25em}
\textbf{Percentil muito alto, inconsistência ou incerteza}: perícia presencial prioritária.
\end{block}
\end{frame}

\begin{frame}[t]{Escore de priorização e limites da automação}
\[
\text{Prioridade de revisão} = P(\text{divergência}) \times \text{impacto do erro} \times \text{incerteza}
\]
\vspace{0.22cm}
\begin{columns}[T,onlytextwidth]
\column{0.315\textwidth}
\begin{block}{Baixo risco}\small Deferimento documental automatizado, quando a norma permitir, com auditoria aleatória.\end{block}
\column{0.315\textwidth}
\begin{block}{Risco intermediário}\small Auditoria documental humana antes da decisão final.\end{block}
\column{0.315\textwidth}
\begin{block}{Alto risco ou incerteza}\small Perícia presencial prioritária ou revisão humana especializada.\end{block}
\end{columns}
\vfill
\FGVCallout{\textbf{Princípio:} o escore automatiza a intensidade do controle; não produz indeferimento automático.}
\end{frame}

\begin{frame}[t]{Validação, custo-efetividade e salvaguardas}
\begin{columns}[T,onlytextwidth]
\column{0.455\textwidth}
\begin{block}{Validação retrospectiva}
\begin{minipage}[t][1.60cm][t]{\linewidth}\raggedright\small
Treinamento com dados históricos; separação entre treino e teste; calibração do risco; avaliação por subgrupos e estabilidade temporal.
\end{minipage}
\end{block}
\column{0.455\textwidth}
\begin{block}{Validação prospectiva}
\begin{minipage}[t][1.60cm][t]{\linewidth}\raggedright\small
Projeto-piloto, auditoria aleatória e comparação entre unidades ou períodos. Avaliação de fila, custo, erro, equidade e taxa de reversão.
\end{minipage}
\end{block}
\end{columns}
\vspace{0.24cm}
\begin{block}{Salvaguardas}\raggedright\small
Sem indeferimento automático; revisão humana nos casos críticos, atípicos ou incertos; auditoria independente; monitoramento de vieses; governança de dados e explicabilidade; certificação digital.
\end{block}
\vspace{0.18cm}
\FGVCallout{\small \textbf{Custo-efetividade:} \(CE=\frac{\text{Custo adicional}}{\text{Casos corretamente direcionados}}\).}

\begin{frame}[t]{Conclusão}
\vspace{0.16cm}
\begin{block}{Tese}\small
O ATESTMED pode ser estruturado como política pública baseada em evidências: a matriz multicritério compara desenhos institucionais, e a estatística organiza a intensidade da revisão.
\end{block}
\vspace{0.32cm}
\begin{block}{Resultado esperado}\small
A capacidade pericial passa a ser empregada onde a revisão humana possui maior valor decisório, sem automatizar indeferimentos e com auditoria permanente.
\end{block}
\vfill
\FGVCallout{\centering\large\textbf{Modelo racionalista + matriz de síntese de avaliação + estatística de risco}\par\vspace{0.22cm}\normalsize Capacidade pericial aplicada onde produz maior valor público.}
\end{frame}
