\begin{frame}[t]{Objeto da análise}
\vspace{0.06cm}
{\large\textbf{O foco é o fluxo pericial como sistema de alocação de capacidade}}\par\vspace{0.18cm}
\begin{itemize}\setlength\itemsep{0.30em}
\item A política envolve entrada, priorização territorial, triagem, análise documental, teleperícia, perícia presencial e mutirões.
\item O objeto da análise é a regra que distribui capacidade e requerimentos entre essas modalidades.
\item A decisão pública relevante é onde atuar, qual instrumento mobilizar e qual intensidade de revisão humana aplicar.
\item O redesenho busca substituir uma lógica predominantemente reativa por critérios explícitos, auditáveis e baseados em evidências.
\end{itemize}
\vspace{0.08cm}
\FGVStatement{A proposta examina o fluxo da Perícia Médica Federal como um problema de alocação territorial e decisória, e não como avaliação isolada de uma ferramenta específica.}
\end{frame}

\begin{frame}[t]{Critério vigente de priorização territorial}
\begin{columns}[T,onlytextwidth]
\column{0.49\textwidth}
\begin{block}{Critério operacional dominante}\small
A escolha de locais para mutirões é orientada principalmente pelo TMEA-PM, o Tempo Médio de Espera para Atendimento em Perícia Médica.
\end{block}
\vspace{0.08cm}
\begin{block}{Fatores político-institucionais}\small
Pressões, articulações territoriais e coalizões políticas também podem influenciar a agenda de mutirões e a visibilidade de determinadas filas.
\end{block}
\column{0.49\textwidth}
\begin{block}{Limite do TMEA-PM}\small
O indicador mede espera média, mas não revela sozinho complexidade da fila, elegibilidade documental, risco de divergência, custo-efetividade ou capacidade local de resposta.
\end{block}
\vspace{0.08cm}
\begin{block}{Redesenho necessário}\small
A priorização territorial deve combinar fila, perfil dos casos, capacidade disponível, potencial de resolução documental, telepericial e presencial.
\end{block}
\end{columns}
\vspace{0.08cm}
\FGVStatement{O problema não é apenas encaminhar melhor cada requerimento; é também decidir onde concentrar a capacidade pericial e quais instrumentos utilizar em cada território.}
\end{frame}

\begin{frame}[t]{Custo variável e qualidade do gasto público}
\begin{columns}[T,onlytextwidth]
\column{0.49\textwidth}
\begin{block}{Atividade extraordinária remunerada}\small
Mutirões envolvem pagamento por perícia realizada. A análise documental também pode ser liberada em finais de semana ou períodos extraordinários, com remuneração específica.
\end{block}
\vspace{0.08cm}
\begin{block}{Risco gerencial}\small
Quando a alocação extraordinária segue apenas TMEA-PM ou pressão político-institucional, o gasto adicional pode não recair sobre os locais e rotas de maior valor resolutivo.
\end{block}
\column{0.49\textwidth}
\begin{block}{Qualidade do gasto}\small
O uso de recursos extraordinários deve considerar custo por caso resolvido, potencial de redução da fila, perfil da demanda e probabilidade de decisão adequada em cada rota.
\end{block}
\vspace{0.08cm}
\begin{block}{Implicação para o fluxo}\small
A organização do fluxo permite decidir quando pagar por mutirão, quando ampliar análise documental, quando usar teleperícia e quando reservar perícia presencial.
\end{block}
\end{columns}
\vspace{0.08cm}
\FGVStatement{A existência de remuneração extraordinária reforça a necessidade de critérios de alocação orientados por evidência e custo-efetividade, e não apenas por estoque de fila ou pressão política.}
\end{frame}

\begin{frame}[t]{Prioridade territorial para capacidade extraordinária}
\small
\begin{block}{Proposta do indicador}
\[
P^{\text{mutirão}}_j=\operatorname{Norm}(\alpha_1TMEA_j+\alpha_2Fila_j+\alpha_3AltoRisco_j+\alpha_4Deficit_j+\alpha_5Iniquidade_j-\alpha_6CustoMarginal_j)
\]
\end{block}
\vspace{0.04cm}
\begin{columns}[T,onlytextwidth]
\column{0.49\textwidth}
\begin{block}{O que acrescenta ao TMEA-PM}\scriptsize
Fila acumulada, volume esperado de casos que demandam revisão, déficit de capacidade ordinária e desigualdade territorial.
\end{block}
\column{0.49\textwidth}
\begin{block}{O que controla no gasto}\scriptsize
Custo marginal da intervenção extraordinária, como mutirão remunerado ou análise documental em fim de semana.
\end{block}
\end{columns}
\vspace{0.06cm}
\FGVStatement{A equação busca transformar a escolha de mutirões em decisão de custo-efetividade e equidade, reduzindo dependência exclusiva do TMEA-PM ou de pressões político-institucionais.}
\end{frame}

\begin{frame}[t]{Arranjo vigente: rotas e respostas operacionais}
\small
\begin{columns}[T,onlytextwidth]
\column{0.49\textwidth}
\begin{block}{Perícia presencial}\small
Rota de maior intensidade clínica e decisória, especialmente relevante quando há complexidade, incerteza ou necessidade de exame direto.
\end{block}
\vspace{0.08cm}
\begin{block}{Teleperícia}\small
Avaliação humana remota, que reduz deslocamentos, mas ainda consome capacidade médico-pericial.
\end{block}
\column{0.49\textwidth}
\begin{block}{Análise documental}\small
Rota documental operacionalizada pelo ATESTMED, com requisitos formais e régua estatística elementar centrada no CID-10.
\end{block}
\vspace{0.08cm}
\begin{block}{Mutirões}\small
Resposta operacional à fila, hoje acionada principalmente por TMEA-PM e por prioridades político-institucionais.
\end{block}
\end{columns}
\vspace{0.08cm}
\FGVStatement{O sistema já combina modalidades. A inovação proposta está na regra de alocação territorial e no roteamento dos casos dentro do fluxo.}
\end{frame}

\begin{frame}[t]{Problema público e redesenho incremental}
\begin{columns}[T,onlytextwidth]
\column{0.49\textwidth}
\begin{block}{Macroalocação}\small
Definir onde concentrar capacidade: mutirão, teleperícia ampliada, reforço presencial ou redistribuição. O TMEA-PM é relevante, mas insuficiente como critério único.
\end{block}
\vspace{0.08cm}
\begin{block}{Microalocação}\small
Definir a rota de cada requerimento: análise documental, auditoria, teleperícia ou perícia presencial, conforme risco, incerteza e contexto clínico-laboral.
\end{block}
\column{0.49\textwidth}
\begin{block}{Efeitos da regra limitada}\small
Filas podem ser tratadas apenas pela espera média, sem distinguir composição dos casos, capacidade resolutiva esperada e custo-efetividade da intervenção.
\end{block}
\vspace{0.08cm}
\begin{block}{Redesenho proposto}\small
Preservar as modalidades atuais, mas combiná-las por critérios explícitos de fila, risco, perfil laboral, potencial documental, capacidade disponível e governança.
\end{block}
\end{columns}
\end{frame}

\begin{frame}[t]{Modelo estatístico: duração, risco e trabalho}
\small
\begin{block}{Estágio 1 - duração esperada do afastamento}\small
\[
Q_{\tau}(D \mid \text{CID, idade, sexo, território, CBO, exigência funcional, histórico laboral})
\]
Produz percentis condicionais: P50, P75, P90 e P95.
\end{block}
\vspace{0.08cm}
\begin{block}{Estágio 2 - risco de divergência ou necessidade de revisão}\small
\[
P(\text{divergência}=1 \mid X)
\]
$X$ inclui duração solicitada, qualidade documental, atividade/CBO, esforço físico, tempo na ocupação, histórico de afastamentos e histórico decisório.
\end{block}
\vspace{0.08cm}
\FGVStatement{O primeiro estágio descreve o padrão esperado; o segundo estima a necessidade de controle adicional. Ambos informam a rota de avaliação.}
\end{frame}

\begin{frame}[t]{Mensuração do impacto e da incerteza}
\small
\begin{block}{Impacto do erro: consequência de escolher a rota inadequada}
\[
I_i=\operatorname{Norm}(\beta_1H_i+\beta_2R_i+\beta_3J_i+\beta_4O_i)
\]
{\scriptsize $H_i$: gravidade funcional/protetiva; $R_i$: retrabalho; $J_i$: risco jurídico-administrativo; $O_i$: custo operacional da correção.}
\end{block}
\vspace{0.06cm}
\begin{block}{Incerteza: confiança limitada na previsão do modelo}
\[
U_i=\operatorname{Norm}(\gamma_1M_i+\gamma_2D_i+\gamma_3V_i+\gamma_4L_i+\gamma_5C_i)
\]
{\scriptsize $M_i$: dados ausentes/inconsistentes; $D_i$: distância da base histórica; $V_i$: desacordo entre modelos; $L_i$: proximidade do limiar; $C_i$: baixa calibração do subgrupo.}
\end{block}
\vspace{0.06cm}
\FGVStatement{Esses índices traduzem informações administrativas, documentais e estatísticas para escala comum de 0 a 1, permitindo comparar casos e acionar revisão humana quando houver alto risco ou alta incerteza.}
\end{frame}

\begin{frame}[t]{Escore de priorização e limites da automação}
\[
\text{Prioridade}_i =
\big[P(\text{divergência}_i)\big]^{w_1}
\times
\big[I_i\big]^{w_2}
\times
\big[U_i\big]^{w_3}
\]
{\scriptsize $I_i$ é impacto do erro, $U_i$ é incerteza da previsão, e $w_1,w_2,w_3$ são pesos institucionais.}
\vspace{0.06cm}
\begin{columns}[T,onlytextwidth]
\column{0.315\textwidth}
\begin{block}{Baixo risco}\small
ATESTMED documental, quando permitido pela norma, com auditoria aleatória posterior.
\end{block}
\column{0.315\textwidth}
\begin{block}{Risco intermediário}\small
Teleperícia ou auditoria documental humana antes da decisão final.
\end{block}
\column{0.315\textwidth}
\begin{block}{Alto risco ou incerteza}\small
Perícia presencial prioritária ou revisão humana especializada.
\end{block}
\end{columns}
\vspace{0.08cm}
\FGVStatement{O escore pondera risco, impacto e incerteza conforme prioridades institucionais. Ele regula a rota de atendimento e não produz indeferimento automático.}
\end{frame}

Requerimento, atestado digital, CID, dias solicitados, atividade/CBO, histórico laboral e dados administrativos autorizados.

Grupo de referência, duração esperada, risco de divergência, impacto do erro e incerteza.

\end{block}
\vspace{0.06cm}

Auditoria independente, registro de divergências, monitoramento de vieses, recalibração e revisão periódica das regras.

\begin{frame}[t]{Aplicação das equações ao caso Rafael}
\small
\begin{columns}[T,onlytextwidth]
\column{0.49\textwidth}
\begin{block}{Entradas ilustrativas do caso}\scriptsize
\textbf{Duração}: 45 dias, estimada como aproximadamente P75 do grupo de referência.\par\vspace{0.16em}
\textbf{Risco}: $P(\text{divergência}\mid X)=0{,}08$.\par\vspace{0.16em}
\textbf{Impacto}: $I_i=0{,}45$, por potencial de retrabalho e necessidade de correção.\par\vspace{0.16em}
\textbf{Incerteza}: $U_i=0{,}18$, com documento íntegro e grupo histórico comparável.
\end{block}
\column{0.49\textwidth}
\begin{block}{Cálculo da prioridade}\scriptsize
Com pesos institucionais ilustrativos $w_1=0{,}50$, $w_2=0{,}30$ e $w_3=0{,}20$:
\[
(0{,}08)^{0{,}50}\times(0{,}45)^{0{,}30}\times(0{,}18)^{0{,}20}\approx0{,}16
\]
Faixas ilustrativas: $<0{,}20$ baixo risco; $0{,}20$--$0{,}40$ risco intermediário; $\geq0{,}40$ alto risco.
\end{block}
\end{columns}
\vspace{0.08cm}
\begin{block}{Rota resultante}\small
Baixa prioridade de revisão: ATESTMED documental, se permitido pela regra vigente, com auditoria aleatória posterior. Se a incerteza ultrapassar o limiar de segurança $U_i\geq U^*$ ou houver inconsistência documental, a rota passa a exigir revisão humana.
\end{block}
\end{frame}
